cmake_minimum_required(VERSION 3.16)
project(orbita_downlink_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
get_filename_component(REPO "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(WS "${REPO}/firmware/components/orbita_ws")
set(CJSON "${REPO}/firmware/managed_components/espressif__cjson/cJSON")
if(NOT EXISTS "${CJSON}/cJSON.c")
    message(FATAL_ERROR "Primero ejecutar idf.py reconfigure para obtener cJSON")
endif()
add_executable(test_ws_downlink ../test_ws_downlink.c
    "${WS}/ws_rx.c" "${WS}/ws_downlink.c" "${CJSON}/cJSON.c")
target_include_directories(test_ws_downlink PRIVATE "${WS}" "${WS}/include" "${CJSON}")
if(MSVC)
    target_compile_options(test_ws_downlink PRIVATE /UNDEBUG /W4)
else()
    target_compile_options(test_ws_downlink PRIVATE -UNDEBUG -Wall -Wextra)
    target_link_libraries(test_ws_downlink PRIVATE m)
endif()
add_test(NAME downlink COMMAND test_ws_downlink)
